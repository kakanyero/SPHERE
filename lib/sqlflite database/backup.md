```dart
import 'package:sqflite/sqflite.dart'; // Imports sqflite plugin for SQLite database operations
import 'package:path/path.dart';        // Imports path package to build file system paths

class DatabaseHelper {
  // Static singleton instance — ensures only one DatabaseHelper exists app-wide
  static final DatabaseHelper instance = DatabaseHelper._init();

  // Holds the single Database object once initialized; nullable until first use
  static Database? _database;

  // Private named constructor — prevents external instantiation
  DatabaseHelper._init();

  // Getter that lazily initializes the database on first access
  Future<Database> get database async {
    if (_database != null) return _database!; // Return existing DB if already open
    _database = await _initDB('app.db');       // Otherwise, initialize it
    return _database!;                         // Return the newly opened DB
  }

  // Builds the full file path and opens (or creates) the database
  Future<Database> _initDB(String filePath) async {
    final dbPath = await getDatabasesPath(); // Gets the device's default DB storage directory
    final path = join(dbPath, filePath);     // Joins directory + filename → e.g. /data/app/app.db

    return await openDatabase(
      path,           // Full path where the .db file will be stored
      version: 1,     // Schema version — increment this when you change the schema
      onCreate: _createDB, // Callback that runs ONLY when DB is created for the first time
    );
  }

  // Called automatically by openDatabase when no DB file exists yet
  Future _createDB(Database db, int version) async {
    await db.execute('''
      CREATE TABLE notes (
        id    INTEGER PRIMARY KEY AUTOINCREMENT, -- Auto-incrementing unique ID for each row
        title TEXT NOT NULL,                     -- Note title, cannot be empty
        body  TEXT NOT NULL                      -- Note body text, cannot be empty
      )
    '''); // Executes raw SQL to create the 'notes' table
  }
}
```

### 2. Model Class

```dart
class Note {
  final int? id;      // Nullable because id is assigned by SQLite on insert, not by us
  final String title; // Title of the note
  final String body;  // Body content of the note

  // Constructor — id is optional since it's auto-generated by the database
  Note({this.id, required this.title, required this.body});

  // Converts a Note object into a Map so sqflite can store it as a DB row
  Map<String, dynamic> toMap() => {
    'id': id,       // Key matches the column name in the CREATE TABLE statement
    'title': title, // Key matches the 'title' column
    'body': body,   // Key matches the 'body' column
  };

  // Factory constructor — builds a Note from a raw DB row (Map returned by sqflite)
  factory Note.fromMap(Map<String, dynamic> map) => Note(
    id: map['id'],       // Reads 'id' column from the row map
    title: map['title'], // Reads 'title' column from the row map
    body: map['body'],   // Reads 'body' column from the row map
  );
}
```

### 3. CRUD Operations

```dart
// CREATE — inserts a new Note row into the database
Future<int> insertNote(Note note) async {
  final db = await instance.database; // Await the singleton DB connection
  return await db.insert(
    'notes',          // Target table name
    note.toMap(),     // Convert Note to Map so sqflite can write it
    conflictAlgorithm: ConflictAlgorithm.replace, // If a conflict on id, replace the old row
  );
}

// READ ALL — retrieves every row from the notes table
Future<List<Note>> fetchAllNotes() async {
  final db = await instance.database;               // Get the open DB connection
  final maps = await db.query(
    'notes',            // Table to query
    orderBy: 'id DESC', // Return newest notes first (highest id = most recently inserted)
  );
  return maps.map((m) => Note.fromMap(m)).toList(); // Convert each raw Map row into a Note object
}

// READ ONE — fetches a single note by its id
Future<Note?> fetchNote(int id) async {
  final db = await instance.database; // Get the open DB connection
  final maps = await db.query(
    'notes',            // Table to query
    where: 'id = ?',    // Filter clause — '?' is a placeholder to prevent SQL injection
    whereArgs: [id],    // Replaces '?' safely with the actual id value
    limit: 1,           // We only expect one result, so stop after the first match
  );
  return maps.isNotEmpty       // Check if any row was found
      ? Note.fromMap(maps.first) // If yes, convert the first row to a Note
      : null;                    // If no match, return null
}

// UPDATE — modifies an existing row matched by id
Future<int> updateNote(Note note) async {
  final db = await instance.database; // Get the open DB connection
  return await db.update(
    'notes',          // Table to update
    note.toMap(),     // New values to write (as a Map)
    where: 'id = ?',  // Only update the row where id matches
    whereArgs: [note.id], // The actual id value to match, passed safely
  );
}

// DELETE — removes a row from the table by id
Future<int> deleteNote(int id) async {
  final db = await instance.database; // Get the open DB connection
  return await db.delete(
    'notes',         // Table to delete from
    where: 'id = ?', // Only delete the row where id matches
    whereArgs: [id], // The actual id value, passed safely to avoid SQL injection
  );
}
```
